Measures included in the Model

Total Sales = CALCULATE(SUM(Fact_Sales[SalesAmount]))

Total Cost = CALCULATE(SUM(Fact_Sales[Cost]))

Total Profit = [Total Sales]-[Total Cost]

Total Orders = CALCULATE(DISTINCTCOUNT(Fact_Sales[OrderNumber]))

Profit Margin = DIVIDE([Total Profit],[Total Sales])

YoY SalesChange % = 
                VAR CurrntSales = [Total Sales]
                VAR LastyearSales = CALCULATE([Total Sales],
                                    DATEADD('Date'[Date],-1,YEAR))
                        RETURN IF(HASONEVALUE(Dim_Products[Category]),DIVIDE(CurrntSales-LastyearSales,LastyearSales),BLANK())

YoY Profit Margin% = 
        VAR ThisYear = [Profit Margin]
        VAR LastYear = CALCULATE([Profit Margin],
                       DATEADD('Date'[Date],-1,YEAR))
            RETURN DIVIDE(ThisYear-LastYear,LastYear)
Yoy Orders = 
        VAR Curryear = [Total Orders]
        VAR Prevyear = CALCULATE([Total Orders],
                       DATEADD('Date'[Date],-1,YEAR))
            RETURN DIVIDE(Curryear-Prevyear,Prevyear)

Total Sales LY = CALCULATE([Total Sales],DATEADD('Date'[Date],-1,YEAR))

Total Profit LY = CALCULATE([Total Profit],DATEADD('Date'[Date],-1,YEAR))
